-- Monthly revenue, VAT and growth by branch
SELECT branch,
       DATE_TRUNC('month', invoice_date)            AS month,
       SUM(net_amount)                               AS revenue,
       SUM(net_amount) * 0.15                        AS vat_due,
       SUM(net_amount) / LAG(SUM(net_amount))
         OVER (PARTITION BY branch ORDER BY DATE_TRUNC('month', invoice_date)) - 1 AS mom_growth
FROM   sales_invoices
WHERE  status = 'posted'
GROUP BY branch, month
ORDER BY branch, month;
# Clean a ledger export and flag unusual expenses
import pandas as pd

df = pd.read_excel("ledger_2025.xlsx", parse_dates=["date"])
df = df.dropna(subset=["amount"]).drop_duplicates("voucher_no")
df["month"] = df["date"].dt.to_period("M")

monthly = df.pivot_table(index="month", columns="account",
                         values="amount", aggfunc="sum").fillna(0)
z = (monthly - monthly.mean()) / monthly.std()
alerts = z[z.abs() > 2].stack()        # months more than 2σ from normal
print(alerts.sort_values(ascending=False).head(10))
-- Power BI measures for a finance dashboard
Revenue = SUM ( Sales[NetAmount] )

Revenue LY = CALCULATE ( [Revenue], SAMEPERIODLASTYEAR ( 'Date'[Date] ) )

Revenue YoY % = DIVIDE ( [Revenue] - [Revenue LY], [Revenue LY] )

Gross Margin % =
DIVIDE ( [Revenue] - SUM ( Sales[Cost] ), [Revenue] )
